cmake_minimum_required(VERSION 3.20)
project(AIG_FastModel_1_0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(aig_fastmodel STATIC fastmodel.cpp fastmodel_c_api.cpp resource_policy.cpp cache_io_probe.cpp)
target_include_directories(aig_fastmodel PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(fastmodel_test ../tests/native_fastmodel_test.cpp)
target_link_libraries(fastmodel_test PRIVATE aig_fastmodel)
enable_testing()
add_test(NAME FastModel_IncrementalCore COMMAND fastmodel_test)

add_executable(fastmodel_resource_test ../tests/native_resource_test.cpp)
target_link_libraries(fastmodel_resource_test PRIVATE aig_fastmodel)
add_test(NAME FastModel_ResourceStability COMMAND fastmodel_resource_test)
